async function callAPI(action, params = {}) {

  const url = new URL(GAS_URL);

  url.searchParams.set('action', action);

  Object.entries(params).forEach(([k, v]) => url.searchParams.set(k, v));

  const res  = await fetch(url.toString());

  const data = await res.json();

  if (data && data.error) throw new Error(data.error);

  return data;

}

function changeMonth(d) {

  state.currentMonth += d;

  if (state.currentMonth > 11) { state.currentMonth = 0; state.currentYear++; }

  if (state.currentMonth < 0)  { state.currentMonth = 11; state.currentYear--; }

  renderCalendar();

}


function renderCalendar() {

  const { currentMonth: m, currentYear: y } = state;

  document.getElementById('cal-month-label').textContent = THAI_MONTHS[m] + ' ' + (y + 543);


  const firstDay = new Date(y, m, 1).getDay();

  const days     = new Date(y, m + 1, 0).getDate();

  const todayStr = toDateStr(new Date());

  const grid = document.getElementById('cal-grid');

  grid.innerHTML = '';


  // empty cells before first day

  for (let i = 0; i < firstDay; i++) {

    const el = document.createElement('div');

    el.className = 'cal-day other-month';

    el.innerHTML = '<div class="cal-day-num"></div>';

    grid.appendChild(el);

  }


  for (let d = 1; d <= days; d++) {

    const ds      = `${y}-${String(m+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;

    const dow     = new Date(y, m, d).getDay();

    const summary = state.calSummary[ds];

    const el      = document.createElement('div');


    let cls = 'cal-day';

    if (dow === 0) cls += ' sunday';

    if (dow === 6) cls += ' saturday';

    if (ds === todayStr) cls += ' today';

    if (summary && summary.total > 0) cls += ' has-data';

    el.className = cls;


    el.innerHTML = `<div class="cal-day-num">${d}</div>`;


    if (summary && summary.total > 0) {

      const dotCls = summary.packed === summary.total ? 'all-done'

                   : summary.packed > 0 ? 'partial' : 'has-orders';

      el.innerHTML += `<div class="cal-dot-wrap"><div class="cal-dot ${dotCls}"></div></div>`;

      el.onclick = () => openDay(ds);

    }

    grid.appendChild(el);

  }

}


function renderUpcomingList() {

  const container = document.getElementById('upcoming-list');

  const today = new Date();

  const items = [];

  for (let i = 0; i < 30; i++) {

    const d = new Date(today); d.setDate(today.getDate() + i);

    const ds = toDateStr(d);

    if (state.calSummary[ds]?.total > 0) items.push({ date: ds, ...state.calSummary[ds] });

  }

  if (!items.length) {

    container.innerHTML = '<div class="empty-state"><div class="empty-icon">📭</div><div class="empty-title">ไม่มีออเดอร์ที่รอดำเนินการ</div><div class="empty-sub">กด ↻ เพื่อ sync ข้อมูลใหม่</div></div>';

    return;

  }

  container.innerHTML = '<div style="font-size:13px;color:var(--muted);font-weight:700;letter-spacing:0.5px;margin-bottom:12px;text-transform:uppercase;">ออเดอร์ที่รอดำเนินการ</div>' +

    items.map(it => {

      const pending = it.total - (it.packed || 0);

      const label   = new Date(it.date + 'T00:00:00').toLocaleDateString('th-TH', { day:'numeric', month:'long', year:'2-digit' });

      return `<div class="upcoming-item" onclick="openDay('${it.date}')">

        <div class="upcoming-num" style="color:${pending>0?'var(--orange)':'var(--green)'}">${pending}</div>

        <div><div class="upcoming-date">${label}</div><div class="upcoming-sub">รอแพ็ค ${pending} / ทั้งหมด ${it.total} ออเดอร์</div></div>

        <div style="margin-left:auto;font-size:22px;color:var(--muted);">›</div>

      </div>`;

    }).join('');

}

function renderDayScreen(dateStr) {

  const label  = new Date(dateStr + 'T00:00:00').toLocaleDateString('th-TH', { weekday:'long', day:'numeric', month:'long', year:'numeric' });

  const total  = state.dayOrders.length;

  const packed = state.dayOrders.filter(o => o.status === 'packed').length;

  document.getElementById('day-title').textContent = label;

  document.getElementById('day-sub').textContent   = `${packed}/${total} แพ็คแล้ว`;

  document.getElementById('day-stats').innerHTML = `

    <div class="stat-card pending"><div class="stat-num">${total-packed}</div><div class="stat-label">รอแพ็ค</div></div>

    <div class="stat-card done"><div class="stat-num">${packed}</div><div class="stat-label">แพ็คแล้ว</div></div>

    <div class="stat-card"><div class="stat-num" style="color:var(--accent)">${total}</div><div class="stat-label">ทั้งหมด</div></div>`;


  const groups = {};

  state.dayOrders.forEach(o => { if (!groups[o.channel]) groups[o.channel] = []; groups[o.channel].push(o); });

  const list = document.getElementById('day-orders-list');

  list.innerHTML = '';

  ['TT','LINE','FB','IG'].forEach(ch => {

    sec.innerHTML = `<div class="channel-header"><span class="channel-badge ch-${ch}">${ch}</span><span>${groups[ch].length} ออเดอร์</span></div>`

      + groups[ch].map(o => renderOrderCard(o)).join('');

    list.appendChild(sec);

  });

}


function renderOrderCard(o) {

  const isPacked   = o.status === 'packed';

  const isMulti    = o.total_qty > 1;

  const cardClass  = isPacked ? 'packed' : isMulti ? 'multi-item' : '';

  const badgeClass = isPacked ? 'done-badge' : isMulti ? 'multi' : 'single';

  const skuList    = [...new Set(o.items.map(i => i.sku))].join(', ');

  return `<div class="order-card ${cardClass}" onclick="openOrder('${o.order_id}')">

    <div class="order-card-inner">

      <div class="order-qty-badge ${badgeClass}">

        <div class="order-qty-num">${isPacked ? '✓' : o.total_qty}</div>

        <div class="order-qty-unit">${isPacked ? 'done' : 'ชิ้น'}</div>

      </div>

      <div class="order-info">

        <div class="order-id">${o.order_id}</div>

        <div class="order-customer">${o.customer}</div>

        <div class="order-items-preview">${skuList}</div>

      <div class="order-status-icon">${isPacked ? '✅' : isMulti ? '⚠️' : '›'}</div>

    </div>

  </div>`;

}

function openOrder(orderId) {

  const order = state.dayOrders.find(o => o.order_id === orderId);

  if (!order) return;


  // Aggregate SKU เดียวกัน

  const agg = {};

  order.items.forEach(item => {

    if (agg[item.sku]) agg[item.sku].qty += item.qty;

    else agg[item.sku] = { ...item };

  });

  order.items     = Object.values(agg);

  order.total_qty = order.items.reduce((s, i) => s + i.qty, 0);


  state.currentOrder = order;

  state.scanCounts   = {};

  order.items.forEach(item => { state.scanCounts[item.sku] = 0; });

  renderPackScreen();

  showScreen('pack');

}


function renderPackScreen() {

  const o = state.currentOrder;

  document.getElementById('pack-title').textContent    = o.customer;

  document.getElementById('pack-sub').textContent      = o.channel + ' · ' + o.order_id;

  document.getElementById('pack-order-id').textContent = o.order_id;

  document.getElementById('pack-customer').textContent = o.customer;

  document.getElementById('pack-meta').textContent     = `${o.channel} · ${o.items.length} รายการ · ${o.total_qty} ชิ้น`;

  renderPackItems();

  updateProgress();

  setScanStatus('พร้อมสแกน', '');

}


function renderPackItems() {

  const o = state.currentOrder;

  document.getElementById('pack-items-list').innerHTML = o.items.map(item => {

    const scanned    = state.scanCounts[item.sku] || 0;

    const pct        = Math.min(100, Math.round(scanned / item.qty * 100));

    const isComplete = scanned >= item.qty;

    const isOver     = scanned > item.qty;

    const cls        = isOver ? 'over' : isComplete ? 'complete' : '';

    return `<div class="pack-item ${cls}">

      <div class="pack-item-header">

        <div class="pack-item-icon">${isComplete ? '📦' : '📋'}</div>

        <div class="pack-item-info">

          <div class="pack-item-sku">${item.sku}</div>

          <div class="pack-item-name">${item.name}</div>

        </div>

        <div class="pack-item-count-wrap">

          <div class="pack-item-big-count ${cls}">

            <span class="scanned">${scanned}</span><span class="sep">/</span><span class="required">${item.qty}</span>

          </div>

          <div class="pack-item-count-label">สแกนแล้ว / ต้องส่ง</div>

        </div>

      </div>

      <div class="pack-item-controls">

        <button class="ctrl-btn reset" onclick="adjustCount('${item.sku}','reset')" ${scanned>0?'':'disabled'}>↺</button>

        <div class="ctrl-spacer"></div>

        <button class="ctrl-btn minus" onclick="adjustCount('${item.sku}','minus')" ${scanned>0?'':'disabled'}>−</button>

        <button class="ctrl-btn plus"  onclick="adjustCount('${item.sku}','plus')">+</button>

      </div>

      <div class="pack-item-bar"><div class="pack-item-bar-fill" style="width:${pct}%"></div></div>

    </div>`;

  }).join('');

}


function adjustCount(sku, action) {

  const c = state.scanCounts[sku] || 0;

  if (action === 'plus')  state.scanCounts[sku] = c + 1;

  if (action === 'minus') state.scanCounts[sku] = Math.max(0, c - 1);

  if (action === 'reset') state.scanCounts[sku] = 0;

  vibrate([30]);

  renderPackItems();

  updateProgress();

}


function updateProgress() {

  const o        = state.currentOrder;

  const scanned  = Object.values(state.scanCounts).reduce((a, b) => a + b, 0);

  const required = o.total_qty;

  const pct      = Math.min(100, Math.round(scanned / required * 100));

  const fill     = document.getElementById('pack-progress-fill');

  fill.style.width = pct + '%';

  fill.className   = 'pack-progress-fill' + (pct === 100 ? ' done' : '');

  const allDone = o.items.every(i => (state.scanCounts[i.sku] || 0) >= i.qty);

  const btn = document.getElementById('btn-complete');

  if (allDone) {

    btn.classList.add('visible'); btn.disabled = false;

    setScanStatus('✓ ครบทุกรายการ — กดยืนยันได้เลย', 'success');

  } else {

    btn.classList.remove('visible');

    setScanStatus(`รอสแกนอีก ${required - scanned} ชิ้น`, '');

  }

}


function setScanStatus(msg, type) {

  const el = document.getElementById('scan-status');

  el.textContent = msg; el.className = 'scan-status ' + (type || '');

}

// ============================================================

async function scanBarcode() {

  // ถ้าอยู่ใน LINE และ LIFF ready → ใช้ scanCodeV2

  if (window.liff && state.liffReady && liff.isInClient()) {

    try {

      if (!liff.isLoggedIn()) {

        liff.login({ redirectUri: window.location.href });

        return;

      }

      const result = await liff.scanCodeV2();

      if (result && result.value) {

        processBarcode(result.value);

      } else {

        showToast('ไม่พบ barcode', 'error');

      }

    } catch(e) {

      // fallback ไป input file

      document.getElementById('camera-input').click();

    }

  } else {

    // browser ปกติ → input file

    document.getElementById('camera-input').click();

  }

}


async function handleCameraInput(event) {

  const file = event.target.files[0];

  if (!file) return;

  showLoading('กำลังอ่าน barcode...');

  try {

    const ZXing  = window.ZXing;

    const hints  = new Map();

    hints.set(ZXing.DecodeHintType.POSSIBLE_FORMATS, [

      ZXing.BarcodeFormat.EAN_13, ZXing.BarcodeFormat.EAN_8,

      ZXing.BarcodeFormat.CODE_128, ZXing.BarcodeFormat.CODE_39,

      ZXing.BarcodeFormat.CODE_93, ZXing.BarcodeFormat.ITF,

      ZXing.BarcodeFormat.UPC_A, ZXing.BarcodeFormat.UPC_E,

      ZXing.BarcodeFormat.CODABAR, ZXing.BarcodeFormat.QR_CODE,

      ZXing.BarcodeFormat.DATA_MATRIX,

    ]);

    hints.set(ZXing.DecodeHintType.TRY_HARDER, true);

    const reader = new ZXing.BrowserMultiFormatReader(hints);

    const imgUrl = URL.createObjectURL(file);

    const result = await reader.decodeFromImageUrl(imgUrl);

    URL.revokeObjectURL(imgUrl);

    processBarcode(result.getText());

  } catch(e) {

    showToast('อ่าน barcode ไม่สำเร็จ — ลองถ่ายใหม่หรือกรอกเอง', 'error');

  } finally {

    hideLoading();

    event.target.value = '';

  }

}


function processBarcode(code) {

  code = String(code).trim();

  if (!code) return;

  const o = state.currentOrder; if (!o) return;


  let matchedSku = null;

  if (state.scanCounts.hasOwnProperty(code)) matchedSku = code;

  if (!matchedSku && state.skuMaster[code]?.sku && state.scanCounts.hasOwnProperty(state.skuMaster[code].sku))

    matchedSku = state.skuMaster[code].sku;


  if (!matchedSku) { setScanStatus('❌ ไม่พบสินค้านี้ใน order', 'error'); vibrate([200, 50, 200]); return; }


  const item = o.items.find(i => i.sku === matchedSku);

  const cur  = state.scanCounts[matchedSku] || 0;

  if (cur >= item.qty) { setScanStatus(`⚠️ ${item.name} — เกินจำนวนแล้ว!`, 'warning'); vibrate([300,50,300,50,300]); return; }


  state.scanCounts[matchedSku] = cur + 1;

  setScanStatus(`✓ ${item.name} — ${state.scanCounts[matchedSku]}/${item.qty}`, 'success');

  vibrate([50]);

  renderPackItems();

  updateProgress();

}


function vibrate(p) { if (navigator.vibrate) navigator.vibrate(p); }

// COMPLETE ORDER

async function confirmComplete() {

  const o = state.currentOrder; if (!o) return;

  if (!o.items.every(i => (state.scanCounts[i.sku] || 0) >= i.qty)) {

    showToast('สินค้ายังไม่ครบ ไม่สามารถยืนยันได้', 'error'); return;

  }

  showLoading('บันทึกสถานะ...');

  try {

    await callAPI('markOrderPacked', { orderId: o.order_id });

    o.status = 'packed';

    const idx = state.dayOrders.findIndex(x => x.order_id === o.order_id);

    if (idx > -1) state.dayOrders[idx].status = 'packed';

    if (state.calSummary[state.currentDay])

      state.calSummary[state.currentDay].packed = (state.calSummary[state.currentDay].packed || 0) + 1;

    showToast('✓ แพ็คเสร็จแล้ว!', 'success');

    setTimeout(() => showScreen('day'), 800);

    renderDayScreen(state.currentDay);

    renderCalendar();

  } catch(e) {

    showToast('บันทึกไม่สำเร็จ', 'error');

  } finally { hideLoading(); }

}
